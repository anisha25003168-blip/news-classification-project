
import streamlit as st
import joblib

# Page settings
st.set_page_config(
    page_title="News Classification",
    page_icon="📰"
)

# Load the trained model
@st.cache_resource
def load_model():
    model = joblib.load("news_classifier.pkl")
    tfidf = joblib.load("tfidf_vectorizer.pkl")
    return model, tfidf

model, tfidf = load_model()

# Website title
st.title("News Classification System")
st.write("Enter a news headline to predict its category.")

# Input box
headline = st.text_area("Enter your news headline:")

# Prediction button
if st.button("Predict Category"):
    if headline.strip():
        features = tfidf.transform([headline])
        prediction = model.predict(features)[0]
        st.success(f"Predicted category: {prediction}")
    else:
        st.warning("Please enter a headline.")
